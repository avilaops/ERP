import { workflow, trigger, node, newCredential, sticky } from '@n8n/workflow-sdk';
const manual = trigger({type:'n8n-nodes-base.manualTrigger',version:1,config:{name:'Revisar biblioteca manualmente',parameters:{},position:[0,0]},output:[{}]});
const daily = trigger({type:'n8n-nodes-base.scheduleTrigger',version:1.4,config:{name:'Diariamente às três',parameters:{rule:{interval:[{field:'days',daysInterval:1,triggerAtHour:3,triggerAtMinute:0}]}},position:[0,180]},output:[{}]});
const enqueue = node({type:'n8n-nodes-base.httpRequest',version:4.5,config:{name:'Enfileirar revisitação diária',parameters:{method:'POST',url:'https://erp.avilaops.com/api/v1/documentos/coletas',authentication:'genericCredentialType',genericAuthType:'httpTemplatedCustomAuth',options:{timeout:20000,response:{response:{responseFormat:'json'}}}},credentials:{httpTemplatedCustomAuth:newCredential('ERP Documentos — diretoria (Bearer)')},position:[320,0]},output:[{enfileirados:0,aviso:'Worker processa separadamente'}]});
const progress = node({type:'n8n-nodes-base.httpRequest',version:4.5,config:{name:'Consultar progresso da biblioteca',parameters:{url:'https://erp.avilaops.com/api/v1/documentos/coletas',authentication:'genericCredentialType',genericAuthType:'httpTemplatedCustomAuth',options:{timeout:20000,response:{response:{responseFormat:'json'}}}},credentials:{httpTemplatedCustomAuth:newCredential('ERP Documentos — diretoria (Bearer)')},position:[620,0]},output:[{summary:[],rows:[]}]});
const note = sticky('## DOCUMENTOS NF-e\nRascunho: requer publicação do módulo ERP, armazenamento persistente, worker ativo e credencial Bearer com chave de escrita da diretoria. Não altera os coletores OpenAI/Brave ou Notas. O POST apenas enfileira URLs indexadas há mais de 24h; falhas exigem revisão. Não contém segredo. Configurar timezone America/Sao_Paulo antes de publicar.',[],{color:5});
export default workflow('erp-documentos-oficiais','[Documentos] Revisitar e supervisionar acervo NF-e — ERP').add(manual.to(enqueue).to(progress)).add(daily.to(enqueue)).add(note);
